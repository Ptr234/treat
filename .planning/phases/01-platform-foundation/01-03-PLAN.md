    **treat/frontend/src/app/api/tickets/route.ts** — list and create tickets:
    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { client, serverClient } from '@/lib/sanity-client';
    import { TICKETS_ADMIN_QUERY } from '@/lib/sanity-queries';

    export async function GET(request: NextRequest) {
      try {
        const { searchParams } = new URL(request.url);
        const from = parseInt(searchParams.get('from') || '0');
        const to = parseInt(searchParams.get('to') || '50');

        const tickets = await client.fetch(TICKETS_ADMIN_QUERY, { from, to });
        return NextResponse.json({ success: true, data: tickets });
      } catch (error) {
        console.error('[GET /api/tickets]', error);
        return NextResponse.json({ error: 'Failed to fetch tickets' }, { status: 500 });
      }
    }

    export async function POST(request: NextRequest) {
      try {
        const body = await request.json();
        const { title, description, category, priority = 'medium', contactEmail, contactName, contactPhone, investorNationality, sector, investmentSize } = body;

        if (!title || !description || !category || !contactEmail || !contactName) {
          return NextResponse.json(
            { error: 'Required fields: title, description, category, contactEmail, contactName' },
            { status: 400 }
          );
        }

        // SLA hours per category (from Cloud Functions logic)
        const slaHours: Record<string, number> = {
          general_inquiry: 24, procedure_query: 8, application_support: 4,
          license_delay: 2, complaint: 2, vip: 1,
        };

        const now = new Date();
        const slaDeadlineAt = new Date(now.getTime() + (slaHours[category] || 24) * 60 * 60 * 1000);

        // Generate reference number: UIA-YYYY-NNNN
        const year = now.getFullYear();
        const count = await serverClient.fetch(`count(*[_type == "ticket" && referenceNumber match "UIA-${year}-*"])`);
        const referenceNumber = `UIA-${year}-${String((count as number) + 1).padStart(4, '0')}`;

        const ticket = await serverClient.create({
          _type: 'ticket',
          referenceNumber,
          title,
          description,
          category,
          priority,
          status: 'NEW',
          contactName,
          contactEmail,
          contactPhone: contactPhone || '',
          investorNationality: investorNationality || '',
          sector: sector || '',
          investmentSize: investmentSize || '',
          slaDeadlineHours: slaHours[category] || 24,
          slaDeadlineAt: slaDeadlineAt.toISOString(),
          isEscalated: false,
          createdAt: now.toISOString(),
        });

        return NextResponse.json({
          success: true,
          referenceNumber: ticket.referenceNumber,
          ticketId: ticket._id,
        }, { status: 201 });
      } catch (error) {
        console.error('[POST /api/tickets]', error);
        return NextResponse.json({ error: 'Failed to create ticket' }, { status: 500 });
      }
    }
    ```

    **treat/frontend/src/app/api/tickets/[id]/route.ts** — get ticket by reference number:
    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { client } from '@/lib/sanity-client';
    import { TICKET_BY_REFERENCE_QUERY, TICKET_MESSAGES_QUERY } from '@/lib/sanity-queries';

    export async function GET(
      _request: NextRequest,
      { params }: { params: Promise<{ id: string }> }
    ) {
      try {
        const { id } = await params;
        const ticket = await client.fetch(TICKET_BY_REFERENCE_QUERY, { ref: id });

        if (!ticket) {
          return NextResponse.json({ error: 'Ticket not found' }, { status: 404 });
        }

        const messages = await client.fetch(TICKET_MESSAGES_QUERY, { ticketId: ticket._id });

        return NextResponse.json({ success: true, data: { ...ticket, messages } });
      } catch (error) {
        console.error('[GET /api/tickets/[id]]', error);
        return NextResponse.json({ error: 'Failed to fetch ticket' }, { status: 500 });
      }
    }
    ```

    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { client } from '@/lib/sanity-client';

    export async function GET(
      _request: NextRequest,
      { params }: { params: Promise<{ id: string }> }
    ) {
      try {
        const { id } = await params;
        // Try by _id first, then by slug
        const event = await client.fetch(
          `*[_type == "event" && (_id == $id || slug.current == $id)][0] {
            _id, title, slug, date, endDate, category, description, location,
            registrationUrl, isPublished, image { asset->{ url } },
            resources[] { title, file { asset->{ url } }, url }
          }`,
          { id }
        );

        if (!event) {
          return NextResponse.json({ error: 'Event not found' }, { status: 404 });
        }

        return NextResponse.json({ success: true, data: event });
      } catch (error) {
        console.error('[GET /api/events/[id]]', error);
        return NextResponse.json({ error: 'Failed to fetch event' }, { status: 500 });
      }
    }
    ```

    **treat/frontend/src/app/api/projects/route.ts** — list licensed projects:
    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { client } from '@/lib/sanity-client';
    import { PROJECTS_QUERY, PROJECTS_BY_SECTOR_QUERY, PROJECTS_MAP_QUERY } from '@/lib/sanity-queries';

    export async function GET(request: NextRequest) {
      try {
        const { searchParams } = new URL(request.url);
        const sector = searchParams.get('sector');
        const mapOnly = searchParams.get('mapOnly') === 'true';

        let query = PROJECTS_QUERY;
        let params: Record<string, string> = {};

        if (mapOnly) {
          query = PROJECTS_MAP_QUERY;
        } else if (sector) {
          query = PROJECTS_BY_SECTOR_QUERY;
          params = { sector };
        }

        const projects = await client.fetch(query, params, {
          next: { revalidate: 86400 }, // Cache for 24 hours (static data)
        });

        return NextResponse.json({ success: true, data: projects, total: (projects as unknown[]).length });
      } catch (error) {
        console.error('[GET /api/projects]', error);
        return NextResponse.json({ error: 'Failed to fetch projects' }, { status: 500 });
      }
    }
    ```
